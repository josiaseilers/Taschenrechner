\textbf{Stäbe und Gelenke}\par\medskip
\begin{tragwerk}
  \Knoten{a}{0}{0}\Knoten{b}{2}{0}\Stab{a}{b}\Beschriftung[below=6pt]{$(a)!0.5!(b)$}{\small\texttt{\textbackslash Stab} (Faser rechts)}
  \Knoten{c}{3}{0}\Knoten{d}{5}{0}\Stab[faser=links]{c}{d}\Beschriftung[below=6pt]{$(c)!0.5!(d)$}{\small\texttt{faser=links}}
  \Knoten{e}{6}{0}\Knoten{f}{8}{0}\Bogen{e}{f}{1.5}\Beschriftung[below=6pt]{$(e)!0.5!(f)$}{\small\texttt{\textbackslash Bogen}}
  \Knoten{g}{9}{0}\Knoten{h}{11}{0}\Stab[faser=keine]{g}{h}\Beschriftung[below=6pt]{$(g)!0.5!(h)$}{\small\texttt{faser=keine}}
\end{tragwerk}

\bigskip\textbf{Lager} (Winkel 0, 90, 180, 270; \texttt{halb} bei durchlaufendem Stab)\par\medskip
\begin{tragwerk}
  \foreach \x/\w in {0/0,1.2/90,2.4/180,3.6/270}{\Knoten{p}{\x}{0}\Festlager[\w]{p}}
  \Beschriftung[below=24pt]{1.8,0}{\small\texttt{\textbackslash Festlager}}
  \foreach \x/\w in {5.4/0,6.6/90,7.8/180,9.0/270}{\Knoten{p}{\x}{0}\Loslager[\w]{p}}
  \Beschriftung[below=24pt]{7.2,0}{\small\texttt{\textbackslash Loslager}}
\end{tragwerk}
